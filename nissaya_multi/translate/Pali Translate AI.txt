                    if (candidate && candidate.content?.parts?.[0]?.text) {
                        const parsedData = JSON.parse(candidate.content.parts[0].text);
                        // a page checked again (same number) replaces the older one; the pages stay in the order of the input
                        for (const pg of parsedData.pages || []) { const k = acc.findIndex(x => x.page_number === pg.page_number); if (k >= 0) acc[k] = pg; else acc.push(pg); }
                        qi++;
                        lastSummary = (lastSummary ? lastSummary + '\n' : '') + `[အစု ${qi}/${queue.length}] ` + (parsedData.overall_summary || '');
                        let expected = []; for (let i = 0; i < qi; i++) expected = expected.concat(inputPageNumbers(queue[i]));
                        renderDynamicResults({ pages: acc, overall_summary: lastSummary }, expected, candidate.finishReason);
                        okRun = true; break;
                    } else { throw new Error("Invalid response format"); }
                } catch (error) {
                    retries--;
                    if (NO_RETRY.some(c => String(error.message).startsWith('HTTP ' + c))) retries = 0;
                    if (retries === 0) lastFailed = true;      // every failure is tried again by itself (a 401 may also go away after a while; after 30 failures in a row the pressing stops)
                    if (retries === 0) showError("ဘာသာပြန်ရာတွင် အမှားအယွင်း ရှိနေပါသည် (" + error.message + ")။ ခလုပ်ကို ပြန်နှိပ်၍ ဤအစုကို ထပ်ကြိုးစားပါ။");
                    else { await new Promise(res => setTimeout(res, delay)); delay *= 2; }
                }
            }
            document.getElementById('loadingState').classList.add('hidden');
            document.getElementById('checkBtn').disabled = false;
            updateLabel();
            return okRun;
        }

        // ONE click checks all the remaining sets, one after another, with a pause between them (a stop button ends it; a failed set stops it - a new click continues from that set)
        let running = false, autoStop = false, failRun = 0;      // failRun: failed tries in a row (after 30 the automatic pressing stops)
        const sleep = ms => new Promise(res => setTimeout(res, ms));
        async function analyzeNissaya() {
            if (running) return;
            const auto = document.getElementById('autoRun').checked;
            const gap = Math.max(0, parseInt(document.getElementById('autoGap').value, 10) || 30);
            running = true; autoStop = false;
            document.getElementById('stopBtn').classList.remove('hidden');
            document.getElementById('checkBtn').disabled = true;
            try {
                while (true) {
                    const ok = await analyzeOne();
                    document.getElementById('checkBtn').disabled = true;
                    if (ok) failRun = 0;
                    if (!ok && lastFailed && !autoStop && qi < queue.length && ++failRun <= 30 && document.getElementById('autoRetry').checked) {      // a failed set (not a 401 / 403 / 404): the button is pressed again by itself
                        const rg = Math.max(5, parseInt(document.getElementById('retryGap').value, 10) || 60);
                        retryCount++; showRetry();
                        for (let sec = rg; sec > 0 && !autoStop; sec--) {
                            document.getElementById('checkLabel').innerText = `အစု ${qi + 1}/${queue.length} မအောင် — ${sec}s ပြန်စစ် (${retryCount})`;
                            await sleep(1000);
                        }
                        if (autoStop) break;
                        continue;
                    }
                    if (!ok || !auto || qi >= queue.length || autoStop) break;
                    for (let sec = gap; sec > 0 && !autoStop; sec--) {
                        document.getElementById('checkLabel').innerText = `အစု ${qi}/${queue.length} ✔ — ${sec}s စောင့်`;
                        await sleep(1000);
                    }
                    if (autoStop) break;
                }
            } finally {
                running = false;
                if (queue.length && qi >= queue.length && acc.length && !autoSaved) { autoSaved = true; try { downloadResult(true); } catch (e) {} }
                document.getElementById('stopBtn').classList.add('hidden');
                document.getElementById('checkBtn').disabled = false;
                updateLabel();
            }
        }
        function stopAuto() { autoStop = true; }
        function resetRetry() { retryCount = 0; showRetry(); }

        function buildAllText(data) {
            return (data.pages || []).map(p => `--- စာမျက်နှာ ${p.page_number} ---\n${p.translation}`).join('\n\n');
        }

        function renderDynamicResults(data, expectedPages, finishReason) {
            lastData = data;
            const pagesContainer = document.getElementById('pagesContainer');
            pagesContainer.innerHTML = '';
            const pages = data.pages || [];

            // check: same pages as the input?
            const got = pages.map(p => p.page_number);
            const missing = (expectedPages || []).filter(n => !got.includes(n));
            const extra = got.filter(n => (expectedPages || []).length && !expectedPages.includes(n));
            const warn = [];
            if (missing.length) warn.push('ပြန်မလာသော စာမျက်နှာ: ' + missing.join(', ') + ' — စာမျက်နှာနည်းနည်းဖြင့် ထပ်စစ်ပါ');
            if (extra.length) warn.push('ထည့်မပေးထားသော စာမျက်နှာ ပါလာသည်: ' + extra.join(', '));
            if (finishReason && finishReason !== 'STOP') warn.push('ရလဒ် ကြားဖြတ်ပြတ်သွားနိုင်သည် (' + finishReason + ') — စာမျက်နှာနည်းနည်းဖြင့် ထပ်စစ်ပါ');
            const wb = document.getElementById('warnBox');
            if (warn.length) { wb.innerHTML = warn.map(escapeHtml).join('<br>'); wb.classList.remove('hidden'); } else { wb.classList.add('hidden'); }

            document.getElementById('allText').value = buildAllText(data);
            document.getElementById('allInfo').innerText = `စာမျက်နှာ ${pages.length} ခု — "--- စာမျက်နှာ N ---" ခြားပြီး တစ်ခါတည်း ကူးပါမည်`;

            pages.forEach((page, index) => {
                const textId = `pageText_${index}`;
                const card = document.createElement('div');
                card.className = 'bg-amber-50/60 rounded-xl p-4 border border-amber-200 shadow-sm';
                const li = (arr, icon) => (arr || []).map(i => `<li class="flex items-start gap-1.5"><i class="fa-solid ${icon} mt-1 text-xs"></i><span>${escapeHtml(i)}</span></li>`).join('');
                let block = '';
                if ((page.key_terms || []).length) block += `<div class="mt-3 bg-white/90 p-3 rounded-lg border border-slate-200"><strong class="text-xs text-slate-500 block mb-1">အဓိက ဝေါဟာရ ရွေးချယ်ချက် -</strong><ul class="text-xs text-slate-700 space-y-1">${li(page.key_terms, 'fa-check text-emerald-600')}</ul></div>`;
                if ((page.uncertain || []).length) block += `<div class="mt-2 bg-rose-50 p-3 rounded-lg border border-rose-200"><strong class="text-xs text-rose-700 block mb-1">မသေချာသော နေရာများ (ကိုယ်တိုင်စစ်ပါ) -</strong><ul class="text-xs text-rose-800 space-y-1">${li(page.uncertain, 'fa-question text-rose-600')}</ul></div>`;
                card.innerHTML = `
                    <div class="flex items-center justify-between mb-3 border-b border-amber-200 pb-2">
                        <div class="flex items-center gap-2 font-bold text-amber-900 text-base">
                            <span class="bg-amber-800 text-white rounded-full px-2.5 py-0.5 text-xs font-sans">${index + 1}</span>
                            <span>နိဿယ စာမျက်နှာ ${escapeHtml(String(page.page_number))}</span>
                        </div>
                        <button onclick="copyToClipboard('${textId}')" class="text-xs bg-amber-100 hover:bg-amber-200 text-amber-900 font-semibold px-2.5 py-1 rounded border border-amber-300 flex items-center gap-1"><i class="fa-regular fa-copy"></i> ဤစာမျက်နှာ ကူးမည်</button>
                    </div>
                    <div class="bg-white p-4 rounded-lg border border-slate-200">
                        <div id="${textId}" class="whitespace-pre-wrap leading-relaxed text-slate-800 bg-slate-50 p-3 rounded border border-slate-100 myanmar-text text-base">${escapeHtml(page.translation)}</div>
                    </div>${block}`;
                pagesContainer.appendChild(card);
            });

            if (data.overall_summary) {
                document.getElementById('overallSummary').innerText = data.overall_summary;
                document.getElementById('summarySection').classList.remove('hidden');
            } else document.getElementById('summarySection').classList.add('hidden');

            document.getElementById('resultContent').classList.remove('hidden');
            document.getElementById('statusBadge').classList.remove('hidden');
        }

        function legacyCopy(text) {
            const ta = document.createElement('textarea'); ta.value = text; document.body.appendChild(ta); ta.select();
            try { document.execCommand('copy'); } catch (e) {}
            document.body.removeChild(ta);
        }
        function copyAll() {
            if (!lastData) return;
            const text = buildAllText(lastData), n = (lastData.pages || []).length;
            const done = () => alert("စာမျက်နှာ " + n + " ခုလုံးကို တစ်ခါတည်း ကူးပြီးပါပြီ။ ကိရိယာ၏ ④ ခလုပ်ဖြင့် ထည့်ပါ။");
            if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(text).then(done).catch(() => { legacyCopy(text); done(); });
            else { legacyCopy(text); done(); }
        }
        function copyToClipboard(id) {
            const text = document.getElementById(id)?.innerText;
            if (text) { legacyCopy(text); alert("ဤစာမျက်နှာကို ကူးပြီးပါပြီ။"); }
        }

        }

        }
